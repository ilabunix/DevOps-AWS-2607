WITH
cw AS (
    SELECT
        (SELECT COALESCE(MAX(ecs_cpu.__value__), 0)
         FROM ecs_cpu) AS ecs_cpu,

        (SELECT COALESCE(MAX(ecs_mem.__value__), 0)
         FROM ecs_mem) AS ecs_mem
)

SELECT CASE

    -- RED / CRITICAL
    WHEN MAX(cw.ecs_cpu) >= 95 THEN 2
    WHEN MAX(cw.ecs_mem) >= 95 THEN 2

    -- AMBER / MODERATE
    WHEN MAX(cw.ecs_cpu) >= 80 THEN 1
    WHEN MAX(cw.ecs_mem) >= 80 THEN 1

    -- HEALTHY
    ELSE 0

END AS ecs_health

FROM cw
